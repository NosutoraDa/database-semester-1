MariaDB [db_perusahaan]> create view info_peg as
    -> select i.nip, i.nama, b.nama_bag
    -> from infoprib i, pekerjaan p, bagian b
    -> where i.nip = p.nip and
    -> p.kode_bag = b.kode_bag;

    -> !
